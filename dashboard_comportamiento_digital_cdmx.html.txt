    // Plugin para dibujar porcentajes dentro de los segmentos de la Dona
    const donutLabelsPlugin = {
      id: 'donutLabelsPlugin',
      afterDatasetsDraw(chart) {
        if (chart.config.type !== 'doughnut') return;
        const { ctx } = chart;
        const meta = chart.getDatasetMeta(0);
        const dataset = chart.data.datasets[0];
        const total = dataset.data.reduce((acc, v) => acc + v, 0);
        if (!total) return;

        ctx.save();
        meta.data.forEach((arc, index) => {
          const val = dataset.data[index];
          const pct = (val / total) * 100;
          if (pct < 6) return; // Evitar amontonamiento en segmentos muy pequeños
          const pos = arc.tooltipPosition();
          ctx.fillStyle = '#FFFFFF';
          ctx.font = '700 12px Inter, sans-serif';
          ctx.textAlign = 'center';
          ctx.textBaseline = 'middle';
          ctx.fillText(`${pct.toFixed(1)}%`, pos.x, pos.y);
        });
        ctx.restore();
      }
    };

    // Plugin para dibujar los valores exactos al final de cada barra horizontal (como en Power BI)
    const horizontalBarLabelsPlugin = {
      id: 'horizontalBarLabelsPlugin',
      afterDatasetsDraw(chart) {
        if (chart.config.type !== 'bar' || chart.options.indexAxis !== 'y') return;
        const { ctx } = chart;
        const suffix = chart.options.plugins.customBarSuffix || '';
        const decimals = chart.options.plugins.customBarDecimals ?? 2;

        ctx.save();
        chart.data.datasets.forEach((dataset, dsIndex) => {
          const meta = chart.getDatasetMeta(dsIndex);
          if (meta.hidden) return;
          meta.data.forEach((bar, index) => {
            const rawVal = dataset.data[index];
            if (rawVal === null || rawVal === undefined || isNaN(rawVal)) return;
            const text = `${Number(rawVal).toFixed(decimals)}${suffix}`;

            ctx.font = chart.data.datasets.length > 1
              ? '600 10.5px Inter, sans-serif'
              : '700 11.5px Inter, sans-serif';
            ctx.fillStyle = dsIndex === 0 ? '#F8FAFC' : '#CBD5E1';
            ctx.textAlign = 'left';
            ctx.textBaseline = 'middle';
            ctx.fillText(text, bar.x + 7, bar.y);
          });
        });
        ctx.restore();
      }
    };

    Chart.register(donutLabelsPlugin, horizontalBarLabelsPlugin);

      const sumAbsErr = filtered.reduce((acc, d) => acc + Math.abs(d.pa - d.pr), 0);

      const avgPa = sumPa / n;
      const avgPr = sumPr / n;
      const avgHr = sumHr / n;
      const riskPct = (sumRisk / n) * 100;
      const mae = sumAbsErr / n;
      const samplePct = (n / totalSample) * 100;

      document.getElementById('kpiTotal').textContent = n.toLocaleString('es-MX');
      document.getElementById('kpiTotalSub').textContent = `${samplePct.toFixed(1)}% del total analizado (${totalSample})`;

      document.getElementById('kpiPromedioReal').textContent = avgPa.toFixed(2);
      document.getElementById('kpiPromedioRealSub').textContent = avgPa >= 7.5
        ? "Rendimiento grupal estable (>= 7.5)"
        : "Alerta: media grupal bajo 7.50";

      document.getElementById('kpiPromedioPred').textContent = avgPr.toFixed(2);
      document.getElementById('kpiPromedioPredSub').textContent = `Error absoluto medio (MAE): ${mae.toFixed(2)} pts`;

      document.getElementById('kpiRedes').textContent = `${avgHr.toFixed(2)}`;
      document.getElementById('kpiRedesSub').textContent = `Horas promedio diarias en pantalla`;

      document.getElementById('kpiRiesgoPct').textContent = `${riskPct.toFixed(1)}%`;
      document.getElementById('kpiRiesgoSub').textContent = `${sumRisk} de ${n} alumnos con promedio < 7.5`;
    }

    function updateCharts(filtered) {
      // 1. Actualizar Dona y su leyenda lateral interactiva
      const countsPerfil = PERFILES_ORDER.map(p => filtered.filter(d => d.p === p).length);
      const totalN = filtered.length;
      chartDonut.data.datasets[0].data = countsPerfil;
      chartDonut.update();

      const legendContainer = document.getElementById('donutCustomLegend');
      legendContainer.innerHTML = '';
      PERFILES_ORDER.forEach((perfil, idx) => {
        const count = countsPerfil[idx];
        const pct = totalN ? ((count / totalN) * 100).toFixed(1) : '0.0';
        const isSelected = state.perfil === perfil;
        const btn = document.createElement('button');
        btn.type = 'button';
        btn.className = `w-full text-left p-2.5 rounded-xl border transition flex items-center justify-between cursor-pointer ${
          isSelected
            ? 'bg-[#E63946]/15 border-[#FF4D5A]'
            : 'bg-[#0D1017]/60 border-[#2C3240] hover:border-slate-500'
        }`;
        btn.innerHTML = `
          <div class="flex items-center gap-2 min-w-0">
            <span class="w-3 h-3 rounded-full shrink-0" style="background-color: ${PERFIL_COLORS[perfil]}"></span>
            <span class="text-xs font-semibold text-slate-200 truncate">${perfil}</span>
          </div>
          <div class="text-right shrink-0 ml-2">
            <span class="text-xs font-bold text-white">${pct}%</span>
            <span class="block text-[10px] text-slate-400">${count} est.</span>
          </div>
        `;
        btn.addEventListener('click', () => {
          state.perfil = (state.perfil === perfil) ? "ALL" : perfil;
          document.getElementById('filterPerfil').value = state.perfil;
          updateDashboard();
        });
        legendContainer.appendChild(btn);
      });

      // 2. Actualizar Líneas Mensuales (Ene = 1 ... Dic = 12)
      const monthlyPa = [];
      const monthlyHr = [];
      for (let m = 1; m <= 12; m++) {
        const rows = filtered.filter(d => d.m === m);
        if (rows.length === 0) {
          monthlyPa.push(null);
          monthlyHr.push(null);
        } else {
          monthlyPa.push(rows.reduce((s, r) => s + r.pa, 0) / rows.length);
          monthlyHr.push(rows.reduce((s, r) => s + r.hr, 0) / rows.length);
        }
      }
      chartLine.data.datasets[0].data = monthlyPa;
      chartLine.data.datasets[1].data = monthlyHr;
      chartLine.update();

      // 3. Actualizar Barra 1: Promedio Académico por Alcaldía (ordenado mayor a menor)
      const alcaldiaMap = {};
      filtered.forEach(d => {
        if (!alcaldiaMap[d.a]) alcaldiaMap[d.a] = { sum: 0, count: 0 };
        alcaldiaMap[d.a].sum += d.pa;
        alcaldiaMap[d.a].count += 1;
      });
      const alcaldiaStats = Object.keys(alcaldiaMap)
        .map(a => ({ name: a, avg: alcaldiaMap[a].sum / alcaldiaMap[a].count }))
        .sort((a, b) => b.avg - a.avg);

      chartBarAlcaldia.data.labels = alcaldiaStats.map(x => x.name);
      chartBarAlcaldia.data.datasets[0].data = alcaldiaStats.map(x => x.avg);
      chartBarAlcaldia.data.datasets[0].backgroundColor = alcaldiaStats.map((x, i) =>
        i === 0 ? '#FF4D5A' : (i < 4 ? '#E63946' : '#B91C1C')
      );
      chartBarAlcaldia.update();

      // 4. Actualizar Barra 2: Horas en Redes Sociales por Perfil Conductual (ordenado mayor a menor)
      const perfilStats = PERFILES_ORDER.map(p => {
        const rows = filtered.filter(d => d.p === p);
        const avgHr = rows.length ? rows.reduce((s, r) => s + r.hr, 0) / rows.length : 0;
        return { name: p, avgHr, count: rows.length };
      }).filter(x => x.count > 0 || state.perfil === "ALL")
        .sort((a, b) => b.avgHr - a.avgHr);

      chartBarPerfil.data.labels = perfilStats.map(x => x.name);
      chartBarPerfil.data.datasets[0].data = perfilStats.map(x => x.avgHr);
      chartBarPerfil.data.datasets[0].backgroundColor = perfilStats.map(x => PERFIL_COLORS[x.name]);
      chartBarPerfil.update();

      // 5. Actualizar Barra 3: Promedio Real vs Predicho por Semestre (1° a 9°)
      const semReal = [];
      const semPred = [];
      for (let s = 1; s <= 9; s++) {
        const rows = filtered.filter(d => d.s === s);
        if (rows.length === 0) {
          semReal.push(0);
          semPred.push(0);
        } else {
          semReal.push(rows.reduce((acc, r) => acc + r.pa, 0) / rows.length);
          semPred.push(rows.reduce((acc, r) => acc + r.pr, 0) / rows.length);
        }
      }
      chartBarSemestre.data.datasets[0].data = semReal;
      chartBarSemestre.data.datasets[1].data = semPred;
      chartBarSemestre.update();
    }

    // 6. LÓGICA DEL SIMULADOR DE REGRESIÓN LINEAL MÚLTIPLE
    // =========================================================================
    function calculateSimulation() {
      const hr = parseFloat(document.getElementById('simRedes').value);
      const hp = parseFloat(document.getElementById('simProd').value);
      const ho = parseFloat(document.getElementById('simOcio').value);
      const hn = parseFloat(document.getElementById('simNoct').value);
      const d = parseFloat(document.getElementById('simDesb').value);

      document.getElementById('valSimRedes').textContent = `${hr.toFixed(2)} hrs`;
      document.getElementById('valSimProd').textContent = `${hp.toFixed(2)} hrs`;
      document.getElementById('valSimOcio').textContent = `${ho.toFixed(2)} hrs`;
      document.getElementById('valSimNoct').textContent = `${hn.toFixed(2)} hrs`;
      document.getElementById('valSimDesb').textContent = `${Math.round(d)} desbloqueos`;

      // Ecuación exacta de Regresión Lineal Múltiple:
      // Promedio = 8.2882 - 0.2649*Redes + 0.3671*Productividad - 0.1531*Ocio - 0.2634*Nocturno - 0.0011*Desbloqueos
      const rawPred = 8.2882 - (0.2649 * hr) + (0.3671 * hp) - (0.1531 * ho) - (0.2634 * hn) - (0.0011 * d);
      const clampedPred = Math.max(0, Math.min(10, rawPred));

      document.getElementById('simResultScore').textContent = clampedPred.toFixed(2);
      document.getElementById('simRawNote').textContent = `Valor exacto de ecuación: ${rawPred.toFixed(2)} pts`;

      const riskBadge = document.getElementById('simRiskBadge');
      if (clampedPred < 7.5) {
        riskBadge.textContent = "Con Riesgo (< 7.5)";
        riskBadge.className = "inline-block mt-1 px-2.5 py-1 rounded-lg text-xs font-bold bg-[#E63946]/20 text-[#FF4D5A] border border-[#E63946]/40";
      } else {
        riskBadge.textContent = "Estable (>= 7.5)";
        riskBadge.className = "inline-block mt-1 px-2.5 py-1 rounded-lg text-xs font-bold bg-emerald-500/15 text-emerald-400 border border-emerald-500/40";
      }

      // Asignación al centroide K-Means más cercano (basado en medias empíricas de los 3 clústeres)
      const centroids = [
        { name: "Productivos", hr: 1.63, hp: 3.92, ho: 1.25, hn: 0.54, d: 58 },
        { name: "Consumidores de Ocio", hr: 3.45, hp: 2.13, ho: 3.12, hn: 1.38, d: 96 },
        { name: "Procrastinadores", hr: 5.57, hp: 1.08, ho: 3.58, hn: 2.52, d: 141 }
      ];
      let bestCluster = centroids[0].name;
      let minDist = Infinity;
      centroids.forEach(c => {
        const dist =
          Math.pow((hr - c.hr) / 1.5, 2) +
          Math.pow((hp - c.hp) / 1.2, 2) +
          Math.pow((ho - c.ho) / 1.1, 2) +
          Math.pow((hn - c.hn) / 0.8, 2) +
          Math.pow((d - c.d) / 35, 2);
        if (dist < minDist) {
          minDist = dist;
          bestCluster = c.name;
        }
      });
      document.getElementById('simClusterBadge').textContent = bestCluster;

      // Desglose de aportes
      const posContrib = 0.3671 * hp;
      const negContrib = -(0.2649 * hr + 0.1531 * ho + 0.2634 * hn + 0.0011 * d);
      document.getElementById('simContribPos').textContent = `+${posContrib.toFixed(2)} pts`;
      document.getElementById('simContribNeg').textContent = `${negContrib.toFixed(2)} pts`;
    }

        document.getElementById('simNoct').value = 2.7;

      });

        document.getElementById(id).addEventListener('input', calculateSimulation);
      });
    }
