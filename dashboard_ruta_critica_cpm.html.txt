    // =========================================================================
    // 2. MOTOR MATEMÁTICO CPM (FORWARD PASS & BACKWARD PASS EN TIEMPO REAL)
    // =========================================================================
    function calculateCPM(taskList) {
      const map = {};
      taskList.forEach(t => {
        t.sucesoras = [];
        t.ES = 0;
        t.EF = 0;
        t.LS = 0;
        t.LF = 0;
        t.holgura = 0;
        t.esCritica = false;
        map[t.id] = t;
      });

      // Construir lista de sucesoras
      taskList.forEach(t => {
        t.predecesoras.forEach(predId => {
          if (map[predId]) {
            map[predId].sucesoras.push(t.id);
          }
        });
      });

      // Ordenamiento topológico (Kahn) para garantizar evaluación segura en cualquier grafo DAG
      const inDegree = {};
      taskList.forEach(t => { inDegree[t.id] = t.predecesoras.length; });
      const queue = taskList.filter(t => inDegree[t.id] === 0).map(t => t.id);
      const topoOrder = [];

      while (queue.length > 0) {
        const currId = queue.shift();
        topoOrder.push(currId);
        map[currId].sucesoras.forEach(succId => {
          inDegree[succId]--;
          if (inDegree[succId] === 0) queue.push(succId);
        });
      }

      // FORWARD PASS: ES = max(EF de predecesoras), EF = ES + duración
      topoOrder.forEach(id => {
        const t = map[id];
        if (t.predecesoras.length === 0) {
          t.ES = 0;
        } else {
          t.ES = Math.max(...t.predecesoras.map(pid => map[pid].EF));
        }
        t.EF = t.ES + t.duracion;
      });

      const totalDuration = Math.max(...taskList.map(t => t.EF), 0);

      // BACKWARD PASS: LF = min(LS de sucesoras), LS = LF - duración, Holgura H = LS - ES
      [...topoOrder].reverse().forEach(id => {
        const t = map[id];
        if (t.sucesoras.length === 0) {
          t.LF = totalDuration;
        } else {
          t.LF = Math.min(...t.sucesoras.map(sid => map[sid].LS));
        }
        t.LS = t.LF - t.duracion;
        t.holgura = t.LS - t.ES;
        t.esCritica = (t.holgura === 0);
      });

      // Reconstruir secuencia(s) de Ruta Crítica mediante DFS sobre aristas críticas ajustadas
      const criticalPaths = [];
      const startNodes = taskList.filter(t => t.predecesoras.length === 0 && t.esCritica);

      function dfsCritical(curr, path) {
        const nextPath = [...path, curr.id];
        if (curr.sucesoras.length === 0 && curr.EF === totalDuration) {
          criticalPaths.push(nextPath);
          return;
        }
        curr.sucesoras.forEach(succId => {
          const succ = map[succId];
          if (succ.esCritica && curr.EF === succ.ES) {
            dfsCritical(succ, nextPath);
          }
        });
      }

      startNodes.forEach(sn => dfsCritical(sn, []));

      return {
        totalDuration,
        criticalPaths,
        taskMap: map
      };
    }

          <filter id="glowRed" x="-20%" y="-20%" width="140%" height="140%">
            <feGaussianBlur stdDeviation="3.5" result="blur" />
            <feComposite in="SourceGraphic" in2="blur" operator="over" />
          </filter>
        </defs>

        <!-- Etiquetas de las 3 Ramas Paralelas -->
        <text x="598" y="14" text-anchor="middle" fill="#64748B" font-size="10" font-weight="700" letter-spacing="0.06em">
          RAMA ESTADÍSTICA (PERSONA 2)
        </text>
        <text x="598" y="155" text-anchor="middle" fill="#64748B" font-size="10" font-weight="700" letter-spacing="0.06em">
          RAMA CLUSTERING (PERSONA 3)
        </text>
        <text x="598" y="296" text-anchor="middle" fill="#64748B" font-size="10" font-weight="700" letter-spacing="0.06em">
          RAMA PREDICTIVA (PERSONA 4)
        </text>
      `;

      // 1. Dibujar Flechas de Precedencia (Curvas Bézier suaves)
      tasks.forEach(target => {
        const dstPos = PERT_LAYOUT[target.id];
        if (!dstPos) return;

        target.predecesoras.forEach(srcId => {
          const src = taskMap[srcId];
          const srcPos = PERT_LAYOUT[srcId];
          if (!src || !srcPos) return;

          const x1 = srcPos.x + NODE_W;
          const y1 = srcPos.y + NODE_H / 2;
          const x2 = dstPos.x - 2;
          const y2 = dstPos.y + NODE_H / 2;

          const isCriticalEdge = src.esCritica && target.esCritica && (src.EF === target.ES);
          const isVisible = filteredIds.has(src.id) || filteredIds.has(target.id);

          const dx = Math.max((x2 - x1) * 0.45, 24);
          const pathD = `M ${x1} ${y1} C ${x1 + dx} ${y1}, ${x2 - dx} ${y2}, ${x2} ${y2}`;

          const strokeColor = !isVisible
            ? "#2A3142"
            : isCriticalEdge
              ? "#FF4D5A"
              : "#0EA5E9";
          const strokeWidth = isCriticalEdge ? 2.6 : 1.7;
          const strokeDash = isCriticalEdge ? "" : 'stroke-dasharray="4 3"';
          const marker = !isVisible
            ? "url(#arrowDim)"
            : isCriticalEdge
              ? "url(#arrowCritical)"
              : "url(#arrowNormal)";

          svgContent += `
            <path d="${pathD}" fill="none" stroke="${strokeColor}" stroke-width="${strokeWidth}" ${strokeDash} marker-end="${marker}" opacity="${isVisible ? 0.92 : 0.3}" />
          `;
        });
      });

      // 2. Dibujar Nodos (10 Tareas)
      tasks.forEach(t => {
        const pos = PERT_LAYOUT[t.id];
        if (!pos) return;

        const isVisible = filteredIds.has(t.id);
        const isSelected = selectedTaskId === t.id;
        const borderColor = isSelected
          ? "#FACC15"
          : t.esCritica
            ? "#FF4D5A"
            : "#38BDF8";
        const headerFill = t.esCritica ? "rgba(239, 68, 68, 0.24)" : "rgba(14, 165, 233, 0.16)";
        const footerFill = t.esCritica ? "rgba(239, 68, 68, 0.14)" : "rgba(15, 23, 42, 0.85)";
        const statusColor =
          t.estado === "Listo" ? "#10B981" :
          t.estado === "En curso" ? "#F59E0B" : "#64748B";
        const personaShort = t.responsable.split(" (")[0];

        svgContent += `
          <g class="pert-node" onclick="selectNode('${t.id}')" opacity="${isVisible ? 1 : 0.26}">
            <!-- Sombra / Resplandor Crítico -->
            ${t.esCritica && isVisible ? `<rect x="${pos.x}" y="${pos.y}" width="${NODE_W}" height="${NODE_H}" rx="11" fill="none" stroke="#EF4444" stroke-width="2" filter="url(#glowRed)" opacity="0.55" />` : ""}

            <!-- Cuerpo Principal del Nodo -->
            <rect x="${pos.x}" y="${pos.y}" width="${NODE_W}" height="${NODE_H}" rx="11" fill="#151A24" stroke="${borderColor}" stroke-width="${isSelected ? 2.5 : 1.6}" />

            <!-- Banda Superior: ES | ID | EF -->
            <rect x="${pos.x + 1}" y="${pos.y + 1}" width="${NODE_W - 2}" height="24" rx="10" fill="${headerFill}" />
            <text x="${pos.x + 10}" y="${pos.y + 16}" fill="#CBD5E1" font-size="9.5" font-weight="600">ES:${t.ES}</text>
            <text x="${pos.x + NODE_W / 2}" y="${pos.y + 16.5}" text-anchor="middle" fill="#FFFFFF" font-size="11" font-weight="800">${t.id}</text>
            <text x="${pos.x + NODE_W - 10}" y="${pos.y + 16}" text-anchor="end" fill="#CBD5E1" font-size="9.5" font-weight="600">EF:${t.EF}</text>

            <!-- Punto Estado Jira -->
            <circle cx="${pos.x + NODE_W / 2 + 19}" cy="${pos.y + 13}" r="3.5" fill="${statusColor}" />

            <!-- Centro: Título Corto, Responsable y Duración -->
            <text x="${pos.x + NODE_W / 2}" y="${pos.y + 43}" text-anchor="middle" fill="#F8FAFC" font-size="10" font-weight="700">
              ${pos.shortTitle}
            </text>
            <text x="${pos.x + NODE_W / 2}" y="${pos.y + 59}" text-anchor="middle" fill="#94A3B8" font-size="9.5" font-weight="500">
              ${personaShort} · <tspan fill="#FFFFFF" font-weight="700">${t.duracion}d</tspan>
            </text>

            <!-- Línea divisoria inferior -->
            <line x1="${pos.x + 1}" y1="${pos.y + 74}" x2="${pos.x + NODE_W - 1}" y2="${pos.y + 74}" stroke="#2A3142" stroke-width="1" />

            <!-- Banda Inferior: LS | Holgura | LF -->
            <rect x="${pos.x + 1}" y="${pos.y + 75}" width="${NODE_W - 2}" height="30" rx="10" fill="${footerFill}" />
            <text x="${pos.x + 10}" y="${pos.y + 93}" fill="#94A3B8" font-size="9.5" font-weight="600">LS:${t.LS}</text>
            <text x="${pos.x + NODE_W / 2}" y="${pos.y + 93.5}" text-anchor="middle" fill="${t.esCritica ? "#FF4D5A" : "#38BDF8"}" font-size="10" font-weight="800">
              H:${t.holgura}d
            </text>
            <text x="${pos.x + NODE_W - 10}" y="${pos.y + 93}" text-anchor="end" fill="#94A3B8" font-size="9.5" font-weight="600">LF:${t.LF}</text>
          </g>
        `;
      });

      svg.innerHTML = svgContent;
    }

    // 8. RENDERIZADO DE LA TABLA COMPLETA DE CÁLCULO CPM Y CONTROL JIRA
    // =========================================================================
    function renderTable(filteredTasks) {
      const tbody = document.getElementById("cpmTableBody");

      if (filteredTasks.length === 0) {
        tbody.innerHTML = `
          <tr>
            <td colspan="12" class="py-8 text-center text-xs text-slate-400">
              No hay registros que mostrar con los filtros actuales.
            </td>
          </tr>
        `;
        return;
      }

      tbody.innerHTML = filteredTasks.map(t => {
        const isSelected = selectedTaskId === t.id;
        const predsText = t.predecesoras.length ? t.predecesoras.join(", ") : "—";

        const statusSelectStyle =
          t.estado === "Listo"
            ? "border-[#10B981]/50 text-[#10B981] bg-[#10B981]/10"
            : t.estado === "En curso"
              ? "border-[#F59E0B]/50 text-[#F59E0B] bg-[#F59E0B]/10"
              : "border-[#64748B]/50 text-slate-300 bg-[#64748B]/10";

        const classificationBadge = t.esCritica
          ? `<span class="inline-flex items-center gap-1 px-2.5 py-1 rounded-lg text-[11px] font-extrabold bg-[#EF4444]/20 text-[#FF4D5A] border border-[#EF4444]/45">
               <span class="w-1.5 h-1.5 rounded-full bg-[#FF4D5A]"></span> RUTA CRÍTICA
             </span>`
          : `<span class="inline-flex items-center gap-1 px-2.5 py-1 rounded-lg text-[11px] font-bold bg-[#0EA5E9]/15 text-[#38BDF8] border border-[#0EA5E9]/40">
               <span class="w-1.5 h-1.5 rounded-full bg-[#38BDF8]"></span> CON HOLGURA
             </span>`;

        return `
          <tr onclick="selectNode('${t.id}')" class="transition cursor-pointer ${isSelected ? "bg-[#1C2333]" : "hover:bg-[#1A202E]/70"}">
            <td class="py-3 px-3.5 font-extrabold ${t.esCritica ? "text-[#FF4D5A]" : "text-[#38BDF8]"}">${t.id}</td>
            <td class="py-3 px-3.5">
              <div class="font-semibold text-slate-100">${t.nombre}</div>
              <div class="text-[11px] text-slate-400">${t.fase}</div>
            </td>
            <td class="py-3 px-3.5 text-xs text-slate-300 whitespace-nowrap">${t.responsable}</td>
            <td class="py-3 px-3 text-center font-mono text-xs text-slate-300">${predsText}</td>
            <td class="py-3 px-3 text-center" onclick="event.stopPropagation()">
              <div class="inline-flex items-center gap-1 bg-[#0D1017] px-1.5 py-1 rounded-lg border border-[#2A3142]">
                <button type="button" onclick="changeDuration('${t.id}', -1)" class="w-5 h-5 rounded hover:bg-[#2A3142] text-slate-300 font-bold text-xs cursor-pointer">−</button>
                <span class="font-bold text-white text-xs w-6 text-center">${t.duracion}d</span>
                <button type="button" onclick="changeDuration('${t.id}', 1)" class="w-5 h-5 rounded hover:bg-[#2A3142] text-slate-300 font-bold text-xs cursor-pointer">+</button>
              </div>
            </td>
            <td class="py-3 px-2.5 text-center font-mono text-xs text-slate-300">${t.ES}</td>
            <td class="py-3 px-2.5 text-center font-mono text-xs text-slate-300">${t.EF}</td>
            <td class="py-3 px-2.5 text-center font-mono text-xs text-slate-300">${t.LS}</td>
            <td class="py-3 px-2.5 text-center font-mono text-xs text-slate-300">${t.LF}</td>
            <td class="py-3 px-3 text-center font-extrabold text-xs ${t.esCritica ? "text-[#FF4D5A]" : "text-[#38BDF8]"}">
              ${t.holgura}d
            </td>
            <td class="py-3 px-3.5" onclick="event.stopPropagation()">
              <select onchange="changeJiraStatus('${t.id}', this.value)"
                class="pbi-select text-xs font-bold py-1.5 px-2.5 cursor-pointer ${statusSelectStyle}">
                <option value="Listo" ${t.estado === "Listo" ? "selected" : ""}>Listo</option>
                <option value="En curso" ${t.estado === "En curso" ? "selected" : ""}>En curso</option>
                <option value="Por hacer" ${t.estado === "Por hacer" ? "selected" : ""}>Por hacer</option>
              </select>
            </td>
            <td class="py-3 px-3.5 text-right whitespace-nowrap">${classificationBadge}</td>
          </tr>
        `;
      }).join("");
    }

    function renderAll() {
      const cpmResult = calculateCPM(tasks);
      const filteredTasks = getFilteredTasks();

      renderKPIs(cpmResult, filteredTasks);
      renderGantt(cpmResult, filteredTasks);
      renderPERTNetwork(cpmResult, filteredTasks);
      renderNodeDetailPanel();
      renderTable(filteredTasks);
    }

        filters.tipoRuta = e.target.value;
        renderAll();
      });
      document.getElementById("filterEstadoJira").addEventListener("change", (e) => {
        filters.estadoJira = e.target.value;
        renderAll();
      });

      renderAll();
    });
